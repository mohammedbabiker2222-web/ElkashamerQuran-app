/* الصلاة القادمة - الرياض */ async function updateNextPrayer() { const element = document.getElementById("nextPrayer"); try { const response = await fetch( "https://api.aladhan.com/v1/timingsByCity?city=Riyadh&country=Saudi%20Arabia&method=4" ); if (!response.ok) { throw new Error("تعذر تحميل المواقيت"); } const result = await response.json(); const timings = result.data.timings; const prayers = [ { key: "Fajr", name: "الفجر" }, { key: "Dhuhr", name: "الظهر" }, { key: "Asr", name: "العصر" }, { key: "Maghrib", name: "المغرب" }, { key: "Isha", name: "العشاء" } ]; const now = new Date(); for (const prayer of prayers) { const [hour, minute] = timings[prayer.key] .split(":") .map(Number); const prayerTime = new Date(); prayerTime.setHours(hour, minute, 0, 0); if (prayerTime > now) { element.textContent = prayer.name + " " + prayerTime.toLocaleTimeString("ar-SA", { hour: "numeric", minute: "2-digit", hour12: true, timeZone: "Asia/Riyadh" }); return; } } const [hour, minute] = timings.Fajr.split(":").map(Number); const tomorrow = new Date(); tomorrow.setDate(tomorrow.getDate() + 1); tomorrow.setHours(hour, minute, 0, 0); element.textContent = "الفجر " + tomorrow.toLocaleTimeString("ar-SA", { hour: "numeric", minute: "2-digit", hour12: true, timeZone: "Asia/Riyadh" }); } catch (error) { element.textContent = "غير متاح"; } } updateNextPrayer(); setInterval(updateNextPrayer, 60000);
